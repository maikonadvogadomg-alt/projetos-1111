Sub TRF1API()
    Dim http As Object: Set http = CreateObject("MSXML2.XMLHTTP")
    http.Open "GET", "https://api-publica.datajud.cnj.jus.br/api_publica_trf1/_search?q=contrato", False
    http.setRequestHeader "Authorization", "Bearer SEU_TOKEN_AQUI"
    http.send
    Range("A1").Value = http.responseText  'JSON com processos
End Sub
